
    }

  }

  return null;

}

function getEvidenceUrl(

  jobId: string | undefined,

  evidence: Record<string, any> | null | undefined

): string | null {

  if (!jobId || !evidence) {

    return null;

  }

  const rawPath =

    evidence.image_path ||

    evidence.path ||

    evidence.file_path ||

    evidence.filename;

  if (!rawPath) {

    return null;

  }

  const normalized = String(rawPath).replace(/\\/g, "/");

  const filename = normalized.split("/").pop();

  if (!filename) {

    return null;

  }

  return `http://127.0.0.1:8000/evidence/${encodeURIComponent(

    jobId

  )}/${encodeURIComponent(filename)}`;

}
